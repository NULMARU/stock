  key: string;
  labelKey: TranslationKey;
  unitKey: TranslationKey;
  min: number;
  max: number;
  step: number;
  /** Exactly one representative input per display node is shown in basic mode. */
  basic: boolean;
}

export interface NodeGroup {
  node: LiteNodeId;
  labelKey: TranslationKey;
  color: string;
  params: ParamDef[];
}

export function getVisibleParamDefs(mode: ParamMode): ParamDef[] {
  return NODE_GROUPS.flatMap((group) =>
    group.params.filter((param) => mode === 'advanced' || param.basic),
  );
}

export function getParamDef(key: string): ParamDef | undefined {
  return NODE_GROUPS.flatMap((group) => group.params).find((param) => param.key === key);
}

/** Compile-time bridge used by the generic store setter. */
export type LiteParamSetter = <N extends keyof LiteParams>(
  node: N,
  key: keyof LiteParams[N],
  value: number,
) => void;
